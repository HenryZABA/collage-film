cmake_minimum_required(VERSION 3.20)
project(collage_sam2 LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SAM3_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SAM3_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SAM3_SOURCE_DIR}" sam3-lib)
add_executable(collage-sam2 main.cpp)
target_link_libraries(collage-sam2 PRIVATE sam3)
set_target_properties(collage-sam2 PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
